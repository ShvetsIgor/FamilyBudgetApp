const mock = vi.hoisted(() => ({ batches: [] as any[], seq: 0 }));
vi.mock('@/shared/lib/firebase', () => ({ getDb: () => ({}) }));
vi.mock('@/features/recurring/services/recurringExpenseSync', () => ({ restoreRecurringDueFromDeletedExpense: vi.fn().mockResolvedValue(null) }));
vi.mock('firebase/firestore', async (load) => {
 const actual = await load<typeof import('firebase/firestore')>();
 return { ...actual,
  collection: (_:unknown,...p:string[])=>({path:p.join('/')}),
  doc: (db:any,...p:string[])=>({path:p.length?p.join('/'):`${db.path}/id${++mock.seq}`,id:`id${mock.seq}`}),
  query: (...args:unknown[])=>args,
  getDocs: vi.fn().mockResolvedValue({docs:[]}),
  increment: (n:number)=>({increment:n}), serverTimestamp: ()=> 'SERVER_TIME',
  writeBatch: ()=>{const b={set:vi.fn(),update:vi.fn(),delete:vi.fn(),commit:vi.fn().mockResolvedValue(undefined)};mock.batches.push(b);return b;},
 };

import { updateExpense, deleteExpense } from '@/features/expenses/services/expensesService';
import type { SerializableExpense } from '@/shared/types';
const previous: SerializableExpense={id:'e',userId:'u',amount:100,currency:'ILS',categoryId:'c',date:'2026-09-10T12:00:00Z',paymentMethod:'other',tags:['savings'],privacy:'regular',splits:[],isRecurring:true,recurringId:'r',goalId:'g',goalOwnerId:'u',contributionId:'ct',createdAt:'2026-09-10T12:00:00Z',updatedAt:'2026-09-10T12:00:00Z'};
beforeEach(()=>{mock.batches.length=0;});
it('reproduces editing through form input: returned expense loses savings and recurring links',async()=>{
 const updated=await updateExpense({userId:'u',id:'e',amount:150,currency:'ILS',categoryId:'c',date:new Date(previous.date),paymentMethod:'other',tags:['savings'],privacy:'regular',splits:[],previousExpense:previous});
 expect(updated.goalId).toBeUndefined();expect(updated.contributionId).toBeUndefined();expect(updated.recurringId).toBeUndefined();expect(updated.isRecurring).toBe(false);
 expect(mock.batches[0].update.mock.calls[0][1].isRecurring).toBe(false);
});
it('reproduces stale concurrent edits: 100 -> 150 and 100 -> 200 both apply increments',async()=>{
 for(const amount of [150,200])await updateExpense({userId:'u',id:'e',amount,currency:'ILS',categoryId:'c',date:new Date(previous.date),paymentMethod:'other',tags:[],privacy:'regular',splits:[],previousExpense:previous});
 const deltas=mock.batches.map(b=>b.set.mock.calls[0][1].totalExpenses.increment);
 expect(100+deltas.reduce((a,b)=>a+b,0)).toBe(250);
 expect(mock.batches[1].update.mock.calls[0][1].amount).toBe(200);
});
it('reproduces repeat delete: two successful delete requests subtract the same expense twice',async()=>{
 await deleteExpense('u',previous);await deleteExpense('u',previous);
 expect(mock.batches.map(b=>b.set.mock.calls[0][1].totalExpenses.increment)).toEqual([-100,-100]);
});
